```sh

MCIS_DOTNET="$DOTNET_BIN" MCIS_BROWSER_BIN="$CHROMIUM_BIN" python tests_acceptance/ARCHIVEB/browser_real.py
sha256sum -c tests_acceptance/ARCHIVEB/verified-impl-sha256.txt
```
