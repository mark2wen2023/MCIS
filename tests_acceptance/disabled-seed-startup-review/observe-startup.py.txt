import sys
import json


def install_observer(stream):
    # stream must already refer to a new metadata-only log outside the input trees.
    # Never serialize arbitrary event arguments, code constants or file contents.
    def emit(row):
        stream.write(json.dumps(row, ensure_ascii=True) + '\n')
        stream.flush()
    def hook(event, args):
        if event == 'import':
            emit({'event':event, 'module':args[0],
                  'filename':args[1] if isinstance(args[1],str) else None})
        elif event == 'open':
            emit({'event':event, 'path':args[0] if isinstance(args[0],(str,int)) else '<non-path>',
                  'mode':args[1], 'flags':args[2]})
        elif event == 'exec':
            emit({'event':event, 'filename':getattr(args[0],'co_filename',None)})
        elif event == 'compile':
            emit({'event':event, 'filename':args[1]})  # Never log args[0] source.
        elif event in ('subprocess.Popen','os.system','socket.connect'):
            emit({'event':event, 'unexpected':True})  # No command/env/address details.
    sys.addaudithook(hook)
    emit({'event':'baseline-modules', 'modules':sorted(sys.modules)})
    return emit
