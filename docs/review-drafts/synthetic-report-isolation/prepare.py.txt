from pathlib import Path, PurePosixPath
import hashlib
import json
import stat
import subprocess

BASE = '3fed64892327c59902cd6b339d23b1fd2f5f2486'
ALLOWLIST_SHA256 = '9e261e44cc7820724579f7f1fc0f3639b46150bc91f467d299c3022a1cd487cd'
CID = 'CASE-SYNTHETIC-REPORT-ISOLATION'
SEED = {'cases': [{'caseId': CID, 'isFictionalDemo': True, 'sourceRefs': [], 'whatif': {'rows': []}}]}
SAMPLE = b'SYNTHETIC evidence text only; not an accident record.\n'

def need(condition, message):
    if not condition:
        raise ValueError(message)

def sha(raw):
    return hashlib.sha256(raw).hexdigest()

def plain(path, *, file=False):
    p = Path(path)
    need(p.is_absolute() and '..' not in p.parts, 'absolute canonical path required')
    for q in (p, *p.parents):
        need(not q.is_symlink(), 'symlink forbidden')
    if file:
        need(stat.S_ISREG(p.lstat().st_mode), 'regular file required')
    return p

def inventory(path):
    raw = plain(path, file=True).read_bytes()
    need(sha(raw) == ALLOWLIST_SHA256, 'unapproved allowlist')
    result = {}
    for line in raw.decode('utf-8').splitlines():
        digest, name = line.split('\t')
        p = PurePosixPath(name)
        need(not p.is_absolute() and '..' not in p.parts and str(p) == name, 'invalid name')
        need(name not in result, 'duplicate member')
        need(len(digest) == 64 and all(x in '0123456789abcdef' for x in digest), 'invalid digest')
        result[name] = digest
    need(len(result) == 170, 'wrong inventory count')
    return result

def seed_bytes():
    return (json.dumps(SEED, sort_keys=True, separators=(',', ':'))+'\n').encode()

def expected_files(source_map):
    raw = seed_bytes()
    data = 'service/mcis_service/analysis/data/'
    extra = {data+'w12-analysis-seed.json': raw,
             data+'w12-analysis-seed.sha256': (sha(raw)+'  w12-analysis-seed.json\n').encode(),
             'synthetic-fixtures/source.txt': SAMPLE}
    manifest = {'profile': 'synthetic-report-preparation-v1', 'sourceCommit': BASE,
                'caseId': CID, 'seedSha256': sha(raw), 'sampleSha256': sha(SAMPLE),
                'sourceMembers': sorted(source_map), 'reviewOnly': True, 'generationApproved': False}
    extra['preparation.json'] = (json.dumps(manifest, sort_keys=True, separators=(',', ':'))+'\n').encode()
    return extra

def prepare(source, destination, allowlist):
    source, target = plain(source), plain(destination)
    need(source.is_dir() and target.parent.is_dir() and not target.exists(), 'new destination required')
    need(source != target and source not in target.parents and target not in source.parents, 'overlapping roots')
    head = subprocess.check_output(['git','-C',str(source),'rev-parse','HEAD'],text=True).strip()
    need(head == BASE, 'wrong source HEAD')
    members = inventory(allowlist)
    captured = {}
    for name, digest in members.items():
        raw = plain(source/name, file=True).read_bytes()
        need(sha(raw) == digest, 'source mismatch: '+name)
        captured[name] = raw
    captured.update(expected_files(members))
    target.mkdir(mode=0o700)
    for name, raw in captured.items():
        p = target/name
        p.parent.mkdir(parents=True, exist_ok=True)
        with p.open('xb') as out:
            out.write(raw)
    # No recovery, deletion, retry, app, DB, imports or request execution here.
