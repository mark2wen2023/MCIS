def need(condition, message):
    if not condition:
        raise ValueError(message)

def sha(raw):
    return hashlib.sha256(raw).hexdigest()

def plain(path, *, file=False):
    p = Path(path)
    need(p.is_absolute() and '..' not in p.parts, 'canonical absolute path required')
    for q in (p, *p.parents):
        need(not q.is_symlink(), 'symlink forbidden')
    if file:
        need(stat.S_ISREG(p.lstat().st_mode), 'regular file required')
    return p

def exact_tree(root, expected):
    root = plain(root)
    need(root.is_dir(), 'root missing')
    actual = set()
    allowed_dirs = {'.'}
    for name in expected:
        p = Path(name)
        need(not p.is_absolute() and '..' not in p.parts and p.as_posix() == name, 'invalid inventory path')
        allowed_dirs.update(x.as_posix() for x in p.parents)
    for parent, dirs, files in os.walk(root, followlinks=False):
        for name in dirs:
            p = plain(Path(parent)/name)
            need(stat.S_ISDIR(p.lstat().st_mode), 'non-directory')
            need(p.relative_to(root).as_posix() in allowed_dirs, 'extra directory')
        for name in files:
            p = plain(Path(parent)/name, file=True)
            rel = p.relative_to(root).as_posix()
            need(rel in expected, 'extra member: '+rel)
            need(sha(p.read_bytes()) == expected[rel], 'hash mismatch: '+rel)
            actual.add(rel)
    need(actual == set(expected), 'missing files')
